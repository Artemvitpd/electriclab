""" HybridCache FULLSPEED SUITE v2.1 — версия для коммерческих организаций - Поддержка AES-256 и ГОСТ 34.12-2015 (через pygost) - Безопасное управление ключами через окружение - HTTP API для интеграции """ import os import time import hashlib import base64 import threading from typing import List, Optional from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes from cryptography.hazmat.backends import default_backend try: from pygost.gost3412 import GOST34122015 from pygost.gost3413 import ctr except ImportError: GOST34122015 = None try: from quart import Quart, jsonify, request, send_file app = Quart(__name__) except ImportError: app = None # === Глобальные настройки === CACHE_DIR = os.getenv("HYBRIDCACHE_DIR", "C:/HybridCache") SOURCE_DIR = os.getenv("HYBRIDCACHE_SOURCE", "//Server/OfficeDocs") MAX_CACHE_SIZE = int(os.getenv("HYBRIDCACHE_MAXSIZE", 100 * 1024**3)) CACHE_TTL = int(os.getenv("HYBRIDCACHE_TTL", 60 * 60 * 24 * 7)) ENABLE_ENCRYPTION = os.getenv("HYBRIDCACHE_ENCRYPTION", "1") == "1" ENCRYPTION_MODE = os.getenv("HYBRIDCACHE_ENC_MODE", "aes").lower() # === Ключ шифрования (Base64 из окружения) === SECRET_KEY_ENV = os.getenv("HYBRIDCACHE_SECRET_KEY") if not SECRET_KEY_ENV: raise RuntimeError("HYBRIDCACHE_SECRET_KEY не установлен") try: SECRET_KEY = base64.b64decode(SECRET_KEY_ENV) if len(SECRET_KEY) != 32: raise ValueError("Неверная длина ключа (требуется 32 байта)") except Exception as e: raise RuntimeError(f"Ошибка загрузки ключа: {e}") def ensure_cache_dir(): os.makedirs(CACHE_DIR, exist_ok=True) def cache_path(filename): return os.path.join(CACHE_DIR, filename.replace("/", "_").replace("\\", "_").replace("..", "")) def is_expired(filepath): try: return (time.time() - os.path.getmtime(filepath)) > CACHE_TTL except Exception: return True def get_total_cache_size(): total = 0 for f in os.listdir(CACHE_DIR): try: total += os.path.getsize(cache_path(f)) except Exception: pass return total def enforce_cache_limit(): files = [] for f in os.listdir(CACHE_DIR): cp = cache_path(f) try: mtime = os.path.getmtime(cp) size = os.path.getsize(cp) files.append((f, mtime, size)) except Exception: pass total = sum(size for _, _, size in files) if total <= MAX_CACHE_SIZE: return files.sort(key=lambda x: x[1]) for f, _, size in files: if total <= MAX_CACHE_SIZE: break try: os.remove(cache_path(f)) total -= size except Exception: pass # ==== Шифрование AES ==== def _encrypt_aes(data: bytes, key: bytes) -> bytes: iv = os.urandom(16) cipher = Cipher(algorithms.AES(key[:32]), modes.CTR(iv), backend=default_backend()) encryptor = cipher.encryptor() return iv + encryptor.update(data) + encryptor.finalize() def _decrypt_aes(data: bytes, key: bytes) -> bytes: iv = data[:16] ciphertext = data[16:] cipher = Cipher(algorithms.AES(key[:32]), modes.CTR(iv), backend=default_backend()) decryptor = cipher.decryptor() return decryptor.update(ciphertext) + decryptor.finalize() # ==== Шифрование ГОСТ ==== def _encrypt_gost(data: bytes, key: bytes) -> bytes: if not GOST34122015: raise RuntimeError("Для режима ГОСТ установите: pip install pygost") key = hashlib.sha256(key).digest()[:32] cipher = GOST34122015(key) iv = os.urandom(16) encrypted = ctr(cipher.encrypt, data, iv) return iv + encrypted def _decrypt_gost(data: bytes, key: bytes) -> bytes: if not GOST34122015: raise RuntimeError("Для режима ГОСТ установите: pip install pygost") key = hashlib.sha256(key).digest()[:32] cipher = GOST34122015(key) iv = data[:16] ciphertext = data[16:] return ctr(cipher.encrypt, ciphertext, iv) # ==== Единый интерфейс шифрования ==== def encrypt_data(data: bytes) -> bytes: key = SECRET_KEY if ENCRYPTION_MODE == "gost": return _encrypt_gost(data, key) return _encrypt_aes(data, key) def decrypt_data(data: bytes) -> bytes: key = SECRET_KEY if ENCRYPTION_MODE == "gost": return _decrypt_gost(data, key) return _decrypt_aes(data, key) # ==== Основные функции кэширования ==== def get_from_cache(filename: str, src_dir: Optional[str] = None) -> str: ensure_cache_dir() cpath = cache_path(filename) spath = os.path.join(src_dir or SOURCE_DIR, filename) if os.path.exists(cpath) and not is_expired(cpath): os.utime(cpath, None) return cpath enforce_cache_limit() if not os.path.exists(spath): raise FileNotFoundError(f"Источник не найден: {spath}") with open(spath, "rb") as f: raw_data = f.read() if ENABLE_ENCRYPTION: raw_data = encrypt_data(raw_data) with open(cpath, "wb") as f: f.write(raw_data) return cpath def fast_autosave(filename: str, content: bytes) -> str: ensure_cache_dir() cpath = cache_path(filename) data = encrypt_data(content) if ENABLE_ENCRYPTION else content with open(cpath, "wb") as f: f.write(data) os.utime(cpath, None) enforce_cache_limit() return cpath # ==== HTTP API ==== if app: @app.route("/api/cache/<path:filename>") async def api_get_file(filename): if not filename.isalnum(): return jsonify({"error": "Недопустимое имя файла"}), 400 try: path = get_from_cache(filename) with open(path, "rb") as f: raw = f.read() if ENABLE_ENCRYPTION: raw = decrypt_data(raw) temp_path = cache_path(f"tmp_{hashlib.sha256(filename.encode()).hexdigest()}") with open(temp_path, "wb") as f: f.write(raw) resp = await send_file(temp_path, as_attachment=True, attachment_filename=filename) os.remove(temp_path) return resp except Exception as e: return jsonify({"error": str(e)}), 404 @app.route("/api/preload", methods=["POST"]) async def api_preload(): try: data = await request.get_json(force=True) files = [f for f in data.get("files", []) if f.isalnum()] if not files: return jsonify({"error": "Нет допустимых файлов для прогрева"}), 400 threading.Thread(target=preload_files, args=(files,), daemon=True).start() return jsonify({"status": "preloading", "files": files}) except Exception as e: return jsonify({"error": str(e)}), 400 # ==== Режимы ускорения ==== def preload_files(filelist: List[str], src_dir: Optional[str] = None): for fname in filelist: try: get_from_cache(fname, src_dir) print(f"Preloaded: {fname}") except Exception as e: print(f"Preload failed for {fname}: {e}") def game_mode_preload(maps_dir: str = "D:/Games/MyGame/Maps"): if os.path.exists(maps_dir): files = [f for f in os.listdir(maps_dir) if os.path.isfile(os.path.join(maps_dir, f))] preload_files(files, maps_dir) def media_mode_preload(media_catalog: str = "E:/Media"): if os.path.exists(media_catalog): files = [f for f in os.listdir(media_catalog) if os.path.isfile(os.path.join(media_catalog, f))] preload_files(files, media_catalog) def office_mode_preload(doc_list: List[str]): preload_files(doc_list, SOURCE_DIR) # ==== CLI ==== if __name__ == "__main__": import argparse parser = argparse.ArgumentParser(description="HybridCache FULLSPEED SUITE v2.1 (AES + ГОСТ)") parser.add_argument("--mode", choices=["office", "media", "game"], default="office") parser.add_argument("--preload", help="Файлы для прогрева (через запятую)") parser.add_argument("--run-server", action="store_true") parser.add_argument("--file", help="Получить файл из кэша") parser.add_argument("--autosave-test", action="store_true") parser.add_argument("--enc-mode", choices=["aes", "gost"], default=ENCRYPTION_MODE) args = parser.parse_args() ENCRYPTION_MODE = args.enc_mode os.environ["HYBRIDCACHE_ENC_MODE"] = ENCRYPTION_MODE if args.autosave_test: test_file = "test_autosave.tmp" path = fast_autosave(test_file, b"Test autosave content") print(f"Autosave test: {path}") if args.preload: files = [f.strip() for f in args.preload.split(",")] if args.mode == "office": office_mode_preload(files) else: preload_files(files) if args.mode == "game": print("Игровой режим: прогрев ассетов...") game_mode_preload() elif args.mode == "media": print("Медиа режим: прогрев файлов...") media_mode_preload() elif args.mode == "office" and not args.preload: print("Офисный режим: используйте --preload для документов") if args.file: path = get_from_cache(args.file) print(f"Путь к файлу (через кэш): {path}") if args.run_server and app: print(f"Запуск HTTP API на http://127.0.0.1:8090 (режим шифрования: {ENCRYPTION_MODE})") app.run(host="127.0.0.1", port=8090) elif args.run_server and not app: print("HTTP API недоступен: установите quart") elif not any([args.run_server, args.file, args.preload, args.autosave_test]): print(f"HybridCache FULLSPEED SUITE v2.1 готов (режим шифрования: {ENCRYPTION_MODE})") print("Примеры:") print(" python hybridcache_commercial_v2.1.py --mode office --preload budget.xlsx")
